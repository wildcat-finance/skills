record_path=ROOT/'docs/promise-machine/obligation-gates/demonstration-run.json'

old_raw=record_path.read_bytes();old_prose=prose_path.read_bytes();old=json.loads(old_raw)
(OUT/'original-run.json').write_bytes(old_raw);(OUT/'original-evidence.md.txt').write_bytes(old_prose)

changed=[];new=json.loads(old_raw);prose=old_prose
for entry in new['inputs']:
 raw=owner.confined(entry['path']).read_bytes();digest=hashlib.sha256(raw).hexdigest()
 if entry['sha256']==digest and entry['bytes']==len(raw):continue
 assert entry['path'] in {'tests/promise_machine_coverage.json','docs/promise-machine/obligation-gates/evaluation-run.json'}
 before=dict(entry);entry['sha256']=digest;entry['bytes']=len(raw)
 previous=f"- `{entry['path']}`, SHA-256 `{before['sha256']}`, {before['bytes']} bytes".encode()
 following=f"- `{entry['path']}`, SHA-256 `{digest}`, {len(raw)} bytes".encode()
 assert prose.count(previous)==1
 prose=prose.replace(previous,following)
 changed.append({'old':before,'new':entry})
assert len(changed)==2
assert {k:v for k,v in new.items() if k!='inputs'}=={k:v for k,v in old.items() if k!='inputs'}

(OUT/'rebind.json').write_text(json.dumps(result,sort_keys=True,indent=2)+'\n')
for old_name,new_name in [('original-run.json','before-exchange-obligation-demonstration-run.json'),('original-evidence.md.txt','before-exchange-obligation-demonstration-evidence.md.txt'),('rebind.json','obligation-input-rebind-after-exchange.json')]:
